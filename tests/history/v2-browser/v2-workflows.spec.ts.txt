import { test, expect, type Page } from '@playwright/test';
import type { Snapshot } from '../../src/lib/model';
import type { V2Command } from '../../src/lib/v2-domain';
async function login(page: Page) {
  await page.goto('/login');
  await page.getByRole('button', { name: 'Sign in to your center' }).click();
  await expect(page.getByRole('heading', { name: 'Your center, ready for today.' })).toBeVisible();
}
async function state(page: Page): Promise<Snapshot> {
  return await (await page.request.get('/api/state')).json();
}
async function action(page: Page, command: V2Command) {
  const r = await page.request.post('/api/commands', {
    headers: { Origin: 'http://127.0.0.1:3100' },
    data: { kind: 'v2', command, requestId: crypto.randomUUID() },
  });
  expect(r.ok(), await r.text()).toBe(true);
  return (await r.json()).id as string;
}

test('subject settings and family schedule overrides recalculate exact independent proposals', async ({
  page,
}) => {
  await login(page);
  const db = await state(page),
    s = db.students[2],
    cfg = structuredClone(db.operations!.students[s.id]);
  cfg.subjects.Math.enrolled = true;
  cfg.subjects.Reading.enrolled = true;
  cfg.subjects.Math.nextStart = 101;
  cfg.subjects.Math.packetsPerDay = 2;
  cfg.subjects.Math.sheetsPerPacket = 10;
  cfg.subjects.Reading.format = 'CONNECT';
  cfg.recurringDays = [1, 3];
  await action(page, {
    action: 'student-save',
    studentId: s.id,
    name: s.name,
    earlyLearner: s.earlyLearner,
    settings: cfg,
  });
  const id = await action(page, {
    action: 'plan',
    studentId: s.id,
    subject: 'Math',
    fromDate: '2026-10-12',
    carryIds: [],
  });
  await action(page, {
    action: 'schedule-add',
    studentId: s.id,
    familyId: null,
    kind: 'move-visit',
    start: '2026-10-14',
    end: '2026-10-14',
    returnDate: '2026-10-15',
    reason: 'Fictional Wednesday return moved to Thursday',
  });
  await action(page, {
    action: 'plan',
    studentId: s.id,
    subject: 'Reading',
    fromDate: '2026-10-12',
    carryIds: [],
  });
  const latest = await state(page);
  expect(latest.operations!.assignments.find((a) => a.id === id)?.targetSheets).toBe(40);
  expect(latest.operations!.students[s.id].recurringDays).toEqual([1, 3]);
  await page.getByRole('button', { name: 'Homework', exact: true }).click();
  await page.getByLabel('Search homework').fill(s.name);

  await page.getByRole('button', { name: 'History', exact: true }).click();

    name: s.name,

    'ARCHIVED · V' + (previousVersion + 1),
  );

test('two staff clients cannot overwrite stale profile edits; retried commands are idempotent', async ({
  page,
  browser,
}) => {
  await login(page);
  const second = await browser.newContext();
  const other = await second.newPage();
  await login(other);
  const db = await state(page),
    s = db.students[8],
    cfg = db.operations!.students[s.id];
  const command = {
    action: 'student-save',
    studentId: s.id,
    name: s.name,
    earlyLearner: s.earlyLearner,
    settings: { ...cfg, expectedMinutes: 55 },
  };
  const responses = await Promise.all([
    page.request.post('/api/commands', {
      headers: { Origin: 'http://127.0.0.1:3100' },
      data: { kind: 'v2', command },
    }),
    other.request.post('/api/commands', {
      headers: { Origin: 'http://127.0.0.1:3100' },
      data: { kind: 'v2', command: { ...command, settings: { ...cfg, expectedMinutes: 65 } } },
    }),
  ]);
  expect(responses.map((r) => r.status()).sort()).toEqual([200, 409]);
  const requestId = crypto.randomUUID(),
    body = {
      kind: 'v2',
      requestId,
      command: {
        action: 'assistance',
        studentId: s.id,
        reason: 'Fictional busy-day individual support',
      },
    };
  await page.request.post('/api/commands', {
    headers: { Origin: 'http://127.0.0.1:3100' },
    data: body,
  });
  const duplicate = await page.request.post('/api/commands', {
    headers: { Origin: 'http://127.0.0.1:3100' },
    data: body,
  });
  expect((await duplicate.json()).duplicate).toBe(true);
  expect(
    (await state(page)).operations!.assistance.filter(
      (a) => a.studentId === s.id && a.state !== 'resolved',
    ).length,
  ).toBe(1);
  await second.close();
});
